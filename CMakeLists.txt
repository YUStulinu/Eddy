cmake_minimum_required(VERSION 3.24)
project(Eddy VERSION 0.1.0 LANGUAGES CXX)

option(EDDY_CUDA "Build the CUDA solver (needs the CUDA toolkit)" ON)
option(EDDY_TESTS "Build the tests" ON)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

set(EDDY_WITH_CUDA OFF)
if(EDDY_CUDA)
  include(CheckLanguage)
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
      set(CMAKE_CUDA_ARCHITECTURES native) # pass -DCMAKE_CUDA_ARCHITECTURES=75;86;89 for other GPUs
    endif()
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
    set(CMAKE_CUDA_STANDARD 20)
    set(CMAKE_CUDA_STANDARD_REQUIRED ON)
    set(EDDY_WITH_CUDA ON)
    message(STATUS "Eddy: CUDA solver enabled (CUDA ${CUDAToolkit_VERSION})")
  else()
    message(STATUS "Eddy: no CUDA compiler; CPU solver only")
  endif()
endif()

find_package(OpenMP)

add_library(eddy_core STATIC src/solver_cpu.cpp)
target_include_directories(eddy_core PUBLIC src third_party/stb)
if(OpenMP_CXX_FOUND)
  target_link_libraries(eddy_core PUBLIC OpenMP::OpenMP_CXX)
endif()
if(MSVC)
  target_compile_options(eddy_core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4 /fp:fast /arch:AVX2>)
else()
  target_compile_options(eddy_core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall -Wextra -O3 -march=native -ffast-math>)
endif()
if(EDDY_WITH_CUDA)
  target_sources(eddy_core PRIVATE src/solver_gpu.cu)
  target_compile_definitions(eddy_core PUBLIC EDDY_CUDA)
  target_compile_options(eddy_core PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--use_fast_math -lineinfo>)
  target_link_libraries(eddy_core PUBLIC CUDA::cudart)
endif()

add_executable(eddy src/eddy.cpp)
target_link_libraries(eddy PRIVATE eddy_core)

if(EDDY_TESTS)
  enable_testing()
  add_executable(eddy_tests tests/tests.cpp)
  target_link_libraries(eddy_tests PRIVATE eddy_core)
  add_test(NAME eddy_tests COMMAND eddy_tests)
endif()
